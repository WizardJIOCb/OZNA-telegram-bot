=============================================================================
   💾 DATABASE SCHEMA
=============================================================================

TABLE: project_notes

COLUMNS:
  id                  INT (PRIMARY KEY, AUTO_INCREMENT)
  project_id          INT (NOT NULL)
  user_id             BIGINT (NOT NULL)
  user_fio            VARCHAR(255) (NOT NULL)
  department          VARCHAR(50) (NOT NULL)
  context_menu        VARCHAR(50) (NOT NULL)
  stage               VARCHAR(100) (NOT NULL)
  note_text           TEXT (NOT NULL)
  status              ENUM('active','closed') (DEFAULT 'active')
  created_at          TIMESTAMP (DEFAULT CURRENT_TIMESTAMP)
  updated_at          TIMESTAMP (ON UPDATE CURRENT_TIMESTAMP)
  closed_at           TIMESTAMP (NULL)
  closed_by_user_id   BIGINT (NULL)
  closed_by_fio       VARCHAR(255) (NULL)

INDEXES:
  PRIMARY             - Primary key on id
  idx_project_status  - (project_id, status)
  idx_project_stage   - (project_id, stage)
  idx_user            - (user_id)
  idx_created         - (created_at)
  idx_status          - (status)
